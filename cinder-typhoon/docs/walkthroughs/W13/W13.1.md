import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}

d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://contractor-portal.arwc.test/api/appointment-windows/current", headers=h),
    context=ctx).read())
rec = d["record"]
print("record_id:", rec["record_id"])        # APT-CRR-2026-09-18
print("window day:", rec["window"]["day"])   # Thursday
print("inspection:", rec["inspection"])      # INSP-CRR-2026-09-18
