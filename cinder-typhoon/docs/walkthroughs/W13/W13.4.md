Use appointment_id from W13.1 and roster_id/identity from W13.2.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

# Get appointment_id and roster details
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://contractor-portal.arwc.test/api/appointment-windows/current",
                            headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}),
    context=ctx).read())
apt_id = d1["record"]["record_id"]  # APT-CRR-2026-09-18

d2 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://contractor-portal.arwc.test/api/appointment-windows/current/visitors",
                            headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}),
    context=ctx).read())
rec2 = d2["record"]
roster_id = rec2["record_id"]   # ROSTER-CRR-SEP18-R1
identity = rec2["identity"]     # veybridge.tech.204

# Check in
body = json.dumps({
    "appointment_id": apt_id,
    "roster_id": roster_id,
    "identity": identity,
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://contractor-portal.arwc.test/api/check-ins",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("assignment:", d["record"]["assignment"])
print("flag:", d["flag"])                        # FLAG{25ef1438addf15f7}

pathlib.Path("receipts").mkdir(exist_ok=True)
pathlib.Path("receipts/contractor-session").write_text(d["contractor_session"])
```

Flag: `FLAG{25ef1438addf15f7}` in the top-level `flag` field.
