import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}

d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://corporate-identity.arwc.test/api/provisioning-packages/current", headers=h),
    context=ctx).read())
rec = d["record"]
print("record_id:", rec["record_id"])        # START-MIRA-2026
print("employee_number:", rec["employee_number"])  # TMP-4821
print("assignment:", rec["assignment"])      # PLAN-RELIEF-7
