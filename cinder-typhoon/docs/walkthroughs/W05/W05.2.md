import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}

d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://corporate-identity.arwc.test/api/activations/current", headers=h),
    context=ctx).read())
rec = d["record"]
print("record_id:", rec["record_id"])              # ACT-SEP26-R3
print("activation_code:", rec["activation_code"])  # 731944
print("approved_assignment:", rec["approved_assignment"])  # PLAN-RELIEF-3 (note: different from starter)
print("flag:", rec["flag"])                        # FLAG{e85ec4991d28c04d}
