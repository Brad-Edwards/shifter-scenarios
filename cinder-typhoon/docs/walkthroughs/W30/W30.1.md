Requires W29.3 (paper variance observed). Prepare the remote request plan for the operator dispatcher. The workbench holds a fixed dispatcher contract (`CRR-OP-W30-R1`) and accepts the request only when every field matches it exactly and the running controller is project revision 364 in the OT historian.

Submit the exact plan to `/api/request-plans/current` with the process session. No control client is needed for this step.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
proc = pathlib.Path("receipts/process-session").read_text().strip()
h = {"Authorization": "Bearer " + proc, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

plan = {
    "instruction": "CRR-OP-W30-R1", "asset": "AST-CRR-017", "project_revision": 364,
    "delivery": {"schema": "arwc.remote-request/v1", "request_topic": "arwc/crr/commands/operator",
                 "outcome_topic": "arwc/crr/outcomes/operator", "principal": "operator", "qos": 1,
                 "retain": False},
    "request": {"id_strategy": "fresh_uuid_v4_at_execution", "asset": "AST-CRR-017", "mode": "automatic",
                "position_a_pct": 20.0, "position_b_pct": 20.0, "total_flow_m3s": 0.3,
                "expiry_format": "numeric_unix_epoch_seconds_utc", "expiry_seconds": 90},
    "successful_states": ["received", "dispatch_intent", "dispatched", "plc_accepted", "completed"],
    "retry": {"duplicate": "same_uuid_same_payload_returns_recorded_state_without_another_plc_operation",
              "conflict": "same_uuid_changed_payload_is_conflict",
              "uncertain": "inspect_and_reconcile_never_automatic_retry"},
    "restore": {"channel": "hmi", "mode": "manual", "position_a_pct": 20.0, "position_b_pct": 20.0,
                "total_flow_m3s": 0.2, "remote_enabled": False},
}
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://process-view.arwc.test/api/request-plans/current",
        data=json.dumps(plan).encode(), headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # PLAN-CRR-W30-R1
print("flag:", d["flag"])                        # FLAG{86bf26d3fc61d20c}
```
