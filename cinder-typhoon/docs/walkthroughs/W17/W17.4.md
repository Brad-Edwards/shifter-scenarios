    # The interval must be 15–300 seconds and have at least 5 paired samples.
    cur.execute("""
        SELECT c.variable, h.source_time, h.value, h.unit, h.process_quality AS quality
        FROM public.process_history h

          AND h.source_time >= current_timestamp - interval '2 minutes'

conn.close()
