Read the current state of AST-CRR-017 from FUXA (HMI), OPC UA browser, and historian. Record the UTC observation time and each measured value. Submit to the engineering workbench service.

```python
import json, ssl, urllib.request, pathlib, psycopg2, psycopg2.extras, datetime

UTC = datetime.timezone.utc
ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

# Query OT historian for current measured values.
# From OT engineering desktop psql: psql -h 172.29.63.22 -d historian -U reader
# Or use psycopg2 from the connector:
conn = psycopg2.connect(host="172.29.63.22", port=5432, dbname="historian",
    user="reader", password="yuVxToCIL3DxUeklDAbsjKWbAc8rs8pw", connect_timeout=5)
conn.autocommit = True
with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
    cur.execute("""
        SELECT c.variable, h.source_time, h.value, h.unit, h.process_quality AS quality
        FROM public.process_history h
        JOIN public.historian_tag_catalog c ON c.series = h.tag
        WHERE h.asset = 'AST-CRR-017'
          AND h.source_time >= current_timestamp - interval '5 minutes'
        ORDER BY h.source_time DESC, c.variable
    """)
    rows = cur.fetchall()
conn.close()

# Use the most recent reading for each variable.
by_var = {}
for r in rows:
    if r['variable'] not in by_var:
        by_var[r['variable']] = r

obs_time = by_var['LevelM']['source_time'].astimezone(UTC).isoformat()
def val(v): return float(by_var[v]['value'])

body = json.dumps({
    "card": "W17.1",
    "asset": "AST-CRR-017",
    "observed_at": obs_time,
    "measured": {
        "level_m": val("LevelM"),
        "usable_volume_ml": val("UsableVolumeML"),
        "outlet_a_position_pct": val("PositionPct"),
        "outlet_b_position_pct": val("PositionBPct"),
        "outlet_a_flow_m3s": val("FlowM3s"),
        "outlet_b_flow_m3s": val("FlowBM3s"),
        "total_flow_m3s": val("TotalFlowM3s"),
    },
    "accepted_settings": {
        "outlet_a_position_pct": val("RequestedTargetPct"),
        "outlet_b_position_pct": val("RequestedTargetBPct"),
        "total_flow_m3s": val("RequestedFlowM3s"),
    },
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://process-view.arwc.test/api/live-observations/current",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("satisfied:", d["satisfied"])
print("flag:", d["flag"])   # FLAG{33a22dd3dae3a5a9}
```
