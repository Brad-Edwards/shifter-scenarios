From the OT engineering desktop or the connector container with corporate session.

Requires W17.1. Read the operating state (mode, owner, field data, permissives) from FUXA and OPC UA browser. Submit to the engineering workbench service.

```python
import json, ssl, urllib.request, pathlib, psycopg2, psycopg2.extras, datetime

UTC = datetime.timezone.utc
ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

conn = psycopg2.connect(host="172.29.63.22", port=5432, dbname="historian",
    user="reader", password="yuVxToCIL3DxUeklDAbsjKWbAc8rs8pw", connect_timeout=5)
conn.autocommit = True
with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
    cur.execute("""
        SELECT c.variable, h.source_time, h.value, h.process_quality AS quality
        FROM public.process_history h
        JOIN public.historian_tag_catalog c ON c.series = h.tag
        WHERE h.asset = 'AST-CRR-017'
          AND h.source_time >= current_timestamp - interval '5 minutes'
        ORDER BY h.source_time DESC, c.variable
    """)
    rows = cur.fetchall()
conn.close()

by_var = {}
for r in rows:
    if r['variable'] not in by_var:
        by_var[r['variable']] = r

obs_time = by_var['AutoMode']['source_time'].astimezone(UTC).isoformat()
def ival(v): return int(round(float(by_var[v]['value'])))

mode_map = {0: "manual", 1: "automatic_flow"}
owner_map = {0: "unowned", 1: "hmi", 2: "dispatcher"}
io_map = {0: "unavailable", 1: "healthy"}

body = json.dumps({
    "card": "W17.2",
    "asset": "AST-CRR-017",
    "observed_at": obs_time,
    "mode": mode_map.get(ival("AutoMode"), "manual"),
    "owner": owner_map.get(ival("Owner"), "unowned"),
    "field_data": io_map.get(ival("IoHealthy"), "unavailable"),
    "permissive_a": bool(ival("PermissiveA")),
    "permissive_b": bool(ival("PermissiveB")),
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://process-view.arwc.test/api/live-observations/current",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("satisfied:", d["satisfied"])
print("flag:", d["flag"])   # FLAG{a6a4563e676656d3}
```

The service maps the integer historian values to the string labels defined in the contract and validates the submission against the product record.

Flag: `FLAG{a6a4563e676656d3}` in the top-level `flag` field.
