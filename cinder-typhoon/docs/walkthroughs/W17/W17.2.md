Requires W17.1. Read the operating state (mode, owner, field data, permissives)
of AST-CRR-017 from the historian and submit it at a stable observation time.

Same access as W17.1: the OT historian is reachable only from the OT engineering
container (psql); submit from the connector (corporate session, CA, route to
process-view). Pick a stable `observed_at` 90-240 seconds back where every needed
variable has a Good sample within 2 seconds, not the racing latest edge.

```bash
DATA=$(sudo docker exec -e PGPASSWORD=yuVxToCIL3DxUeklDAbsjKWbAc8rs8pw cinder-arwc-ot-engineering \
  psql -h 172.29.63.22 -U reader -d historian -tAc \
  "SELECT json_agg(row_to_json(t)) FROM (SELECT c.variable, h.source_time, h.value, h.process_quality AS quality FROM public.process_history h JOIN public.historian_tag_catalog c ON c.series=h.tag WHERE h.asset='AST-CRR-017' AND h.source_time >= current_timestamp - interval '5 minutes') t")
printf '%s' "$DATA" | sudo docker exec -i --user fieldlink cinder-arwc-connector python3 - <<'PY'
import sys, json, ssl, urllib.request, pathlib, datetime
UTC = datetime.timezone.utc
rows = json.load(sys.stdin)
def pt(s): return datetime.datetime.fromisoformat(s).astimezone(UTC)
NEEDED = ["AutoMode", "Owner", "IoHealthy", "PermissiveA", "PermissiveB"]
good = [r for r in rows if r["quality"] == "Good"]
latest = max(pt(r["source_time"]) for r in rows)
chosen = None
for off in range(90, 241):
    cand = (latest - datetime.timedelta(seconds=off)).replace(microsecond=0)
    vals = {}; ok = True
    for v in NEEDED:
        near = [r for r in good if r["variable"] == v and abs((pt(r["source_time"]) - cand).total_seconds()) <= 2]
        if not near: ok = False; break
        vals[v] = min(near, key=lambda r: abs((pt(r["source_time"]) - cand).total_seconds()))
    if ok: chosen = (cand, vals); break
assert chosen, "no stable second with every variable Good within 2s; widen the scan"
observed, vals = chosen
def ival(v): return int(round(float(vals[v]["value"])))
mode_map = {0: "manual", 1: "automatic_flow"}
owner_map = {0: "unowned", 1: "hmi", 2: "dispatcher"}
io_map = {0: "unavailable", 1: "healthy"}
body = json.dumps({
    "card": "W17.2", "asset": "AST-CRR-017", "observed_at": observed.isoformat(),
    "mode": mode_map[ival("AutoMode")], "owner": owner_map[ival("Owner")],
    "field_data": io_map[ival("IoHealthy")],
    "permissive_a": bool(ival("PermissiveA")), "permissive_b": bool(ival("PermissiveB")),
}).encode()
ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("/var/lib/fieldlink-connector/handover/corporate-session").read_text().strip()
d = json.loads(urllib.request.urlopen(urllib.request.Request(
    "https://process-view.arwc.test/api/live-observations/current", data=body,
    headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"},
    method="POST"), context=ctx).read())
print("satisfied:", d["satisfied"]); print("flag:", d["flag"])
PY
```
