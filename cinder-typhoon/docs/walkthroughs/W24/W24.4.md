## Recover the signer key from two leaked nonces and forge a fresh export

The bundle's signature records leak the high bits of their Ed25519 nonces. Two
records give a linear relation that recovers the signer's private scalar. Forge a
signature over the fresh export request and submit it. A wrong `s`, or replaying
an archived record, is rejected with 409; replaying the forged submission is also
rejected with 409.

```bash
PROCESS=$(cat receipts/process-session)
python3 - > /tmp/w24forged.json <<'PY'
import base64, hashlib, json
FIELD_P = 2**255 - 19
CURVE_D = (-121665 * pow(121666, -1, FIELD_P)) % FIELD_P
SCALAR_L = 2**252 + 27742317777372353535851937790883648493
SQRT_M1 = pow(2, (FIELD_P - 1) // 4, FIELD_P)
def recover_x(y, sign):
    x2 = (y * y - 1) * pow(CURVE_D * y * y + 1, -1, FIELD_P) % FIELD_P
    x = pow(x2, (FIELD_P + 3) // 8, FIELD_P)
    if (x * x - x2) % FIELD_P: x = x * SQRT_M1 % FIELD_P
    if (x & 1) != sign: x = FIELD_P - x
    return x
def point_add(a, b):
    x1, y1 = a; x2, y2 = b; t = CURVE_D * x1 * x2 * y1 * y2 % FIELD_P
    return ((x1 * y2 + y1 * x2) * pow(1 + t, -1, FIELD_P) % FIELD_P,
            (y1 * y2 + x1 * x2) * pow(1 - t, -1, FIELD_P) % FIELD_P)
def point_mul(scalar, point):
    r = (0, 1)
    while scalar:
        if scalar & 1: r = point_add(r, point)
        point = point_add(point, point); scalar >>= 1
    return r
def point_encode(point):
    x, y = point
    return (y | ((x & 1) << 255)).to_bytes(32, "little")
BASE_Y = 4 * pow(5, -1, FIELD_P) % FIELD_P
CURVE_BASE = (recover_x(BASE_Y, 0), BASE_Y)

bundle = json.load(open("/tmp/w24open.json"))["opened"]["bundle"]
first, second = bundle["records"][:2]
public = bytes.fromhex(bundle["signer"]["public_key"])
def h(rec):
    return int.from_bytes(hashlib.sha512(bytes.fromhex(rec["R"]) + public + base64.b64decode(rec["canonical_b64"])).digest(), "little") % SCALAR_L
s1 = int.from_bytes(bytes.fromhex(first["s"]), "little"); s2 = int.from_bytes(bytes.fromhex(second["s"]), "little")
h1, h2 = h(first), h(second)
delta = bundle["recovery_metadata"]["signed_low60_delta"]
high_diff = int(second["nonce_bits_60_251"]) - int(first["nonce_bits_60_251"])
recovered = ((s2 - s1) - (high_diff << 60) - delta) * pow(h2 - h1, -1, SCALAR_L) % SCALAR_L
assert point_encode(point_mul(recovered, CURVE_BASE)) == public, "recovered scalar does not match the public key"
message = base64.b64decode(bundle["fresh_export_request_b64"], validate=True)
nonce = int.from_bytes(hashlib.sha512(b"EXP-CAL-CRR-25/authorization").digest(), "little") % SCALAR_L
r_encoded = point_encode(point_mul(nonce, CURVE_BASE))
challenge = int.from_bytes(hashlib.sha512(r_encoded + public + message).digest(), "little") % SCALAR_L
scalar = (nonce + challenge * recovered) % SCALAR_L
print(json.dumps({"export_request_b64": base64.b64encode(message).decode(), "R": r_encoded.hex(), "s": scalar.to_bytes(32, "little").hex()}))
PY
BODY=$(cat /tmp/w24forged.json)
curl -sk --cacert /tmp/arwc-ca.crt -H "Authorization: Bearer $PROCESS" -H "X-ARWC-Tenant: arwc" \
  -H "Content-Type: application/json" -d "$BODY" -X POST \
  https://process-view.arwc.test/api/reporting-submissions
```
