## Recover the padding with a bounded Bleichenbacher attack

The estimator task endpoint is an RSA padding-class oracle. Each query returns a
rotated padding class; subtract the selector's rotation to read the true PKCS#1
conformance bit. Run a bounded Bleichenbacher search (it converges in under 4096
queries), recover the unlock blob (it begins `ENG1`), then open the bundle.

```bash
PROCESS=$(cat receipts/process-session)
python3 - > /tmp/w24open.json <<'PY'
import base64, json, ssl, urllib.error, urllib.request
BASE = "https://process-view.arwc.test"; MASK = (1 << 64) - 1
token = open("/var/lib/fieldlink-connector/receipts/process-session").read().strip()
m = json.load(open("/tmp/w24model.json"))
ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
headers = {"Authorization": "Bearer " + token, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
def call(path, body=None, method="POST"):
    data = None if body is None else json.dumps(body, separators=(",", ":")).encode()
    req = urllib.request.Request(BASE + path, data=data, headers=headers, method=method)
    try: r = urllib.request.urlopen(req, context=ctx)
    except urllib.error.HTTPError as e: r = e
    return r.status, json.loads(r.read())
_, answer = call("/api/signature-requests", None, "GET")
record = answer["record"]; n = int(record["modulus"], 16); e = record["exponent"]; c = int(record["target_ciphertext"], 16)
bucket = record["epoch_bucket"]; B = 1 << (8 * 254); queries = 0
def select(request_id):
    s0 = (request_id ^ m["request_constant"]) & MASK; y = (bucket ^ m["epoch_constant"]) & MASK
    x = (s0 ^ ((s0 << m["left_shift"]) & MASK)) & MASK
    s1 = (x ^ y ^ (x >> m["request_shift"]) ^ (y >> m["epoch_shift"])) & MASK
    return (s1 + y) & MASK
def oracle(multiplier):
    global queries
    request_id = (0x24A0000000000000 + queries) & MASK; queries += 1
    candidate = c * pow(multiplier, e, n) % n
    status, result = call("/api/estimator-tasks/current", {"operation": "oracle", "request_id": f"{request_id:016x}", "epoch_bucket": bucket, "ciphertext": f"{candidate:0512x}"})
    if status != 200: raise RuntimeError((status, result, queries, multiplier))
    return (("ABCD".index(result["observed_class"]) - (select(request_id) & 3)) & 3) != 0
def ceildiv(a, b): return -(-a // b)
intervals = [(2 * B, 3 * B - 1)]; s = 1
for _ in range(1, 2050):
    if len(intervals) > 1:
        s += 1
        while not oracle(s): s += 1
    else:
        a, b = intervals[0]; r = ceildiv(2 * (b * s - 2 * B), n)
        while True:
            lower = ceildiv(2 * B + r * n, b); upper = (3 * B - 1 + r * n) // a; found = False
            for cand in range(lower, upper + 1):
                if oracle(cand): s = cand; found = True; break
            if found: break
            r += 1
    updated = []
    for a, b in intervals:
        for r in range(ceildiv(a * s - (3 * B - 1), n), (b * s - 2 * B) // n + 1):
            lo = max(a, ceildiv(2 * B + r * n, s)); hi = min(b, (3 * B - 1 + r * n) // s)
            if lo <= hi: updated.append((lo, hi))
    intervals = []
    for cur in sorted(updated):
        if intervals and cur[0] <= intervals[-1][1] + 1: intervals[-1] = (intervals[-1][0], max(intervals[-1][1], cur[1]))
        else: intervals.append(cur)
    if len(intervals) == 1 and intervals[0][0] == intervals[0][1]: break
else: raise RuntimeError("interval did not converge")
encoded = intervals[0][0].to_bytes(256, "big"); sep = encoded.index(0, 2); unlock = encoded[sep + 1:]
assert unlock.startswith(b"ENG1")
status, opened = call("/api/estimator-tasks/current", {"operation": "open", "bundle_id": "DIAG-EVID-CRR-R7", "project_id": "CRR-CTRL-R21", "project_revision": 21, "archive_unlock_b64": base64.b64encode(unlock).decode()})
assert status == 201, (status, opened)
print(json.dumps({"queries": queries, "flag": opened.get("flag"), "opened": opened}))
```
