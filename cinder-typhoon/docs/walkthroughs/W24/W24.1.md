## Read the signature-request contract and recover the selector

The endpoint returns the contract and a selector artifact. Disassemble the
artifact and recover its xorshift-style selector model: two 64-bit constants, one
left shift, and two right shifts. Brute force the model and keep the single one
that reproduces every validation case.

```bash
PROCESS=$(cat receipts/process-session)
PV=https://process-view.arwc.test
curl -sk --cacert /tmp/arwc-ca.crt -H "Authorization: Bearer $PROCESS" -H "X-ARWC-Tenant: arwc" \
  "$PV/api/signature-requests" -o /tmp/w24.json
python3 - <<'PY'
import json, base64, hashlib
d = json.load(open("/tmp/w24.json"))
print("flag:", d["flag"])
art = base64.b64decode(d["selector_artifact_b64"], validate=True)
assert hashlib.sha256(art).hexdigest() == d["record"]["selector_artifact"]["sha256"]
open("/var/lib/fieldlink-connector/selector24", "wb").write(art)
PY
objdump -d --no-show-raw-insn /var/lib/fieldlink-connector/selector24 > /tmp/w24dis.txt
python3 - <<'PY'
import json, re
d = json.load(open("/tmp/w24.json"))
cases = d["record"]["selector_validation_cases"]
dis = open("/tmp/w24dis.txt").read()
mask = (1 << 64) - 1
consts = {int(v, 16) for v in re.findall(r"movabs\s+\$0x([0-9a-f]+)", dis)}
ls_set = {int(v, 16) for v in re.findall(r"shl\s+\$0x([0-9a-f]+)", dis)}
rs_set = {int(v, 16) for v in re.findall(r"shr\s+\$0x([0-9a-f]+)", dis)}
def make(rc, ec, ls, rs, es):
    def f(rid, eb):
        x = (rid ^ rc) & mask; y = (eb ^ ec) & mask
        x = (x ^ ((x << ls) & mask)) & mask
        return ((x ^ y ^ (x >> rs) ^ (y >> es)) + y) & mask
    return f
models = []
for rc in consts:
    for ec in consts - {rc}:
        for ls in ls_set:
            for rs in rs_set:
                for es in rs_set:
                    f = make(rc, ec, ls, rs, es)
                    if all(f(int(str(c["request_id"]), 16), int(c["epoch_bucket"])) & 3 == c["rotation"] for c in cases):
                        models.append({"request_constant": rc, "epoch_constant": ec,
                                       "left_shift": ls, "request_shift": rs, "epoch_shift": es})
assert len(models) == 1, f"expected one model, found {len(models)}"
json.dump(models[0], open("/tmp/w24model.json", "w"))
print("selector model:", models[0])
PY
```

Expected: `flag: FLAG{55f32a8cc75f8721}`. The recovered model is saved to
`/tmp/w24model.json` for W24.2 and W24.3. The flag is in the top-level `flag` field.
