Requires W12.1 (quarantine record). Parse the AQTN1 frame and process sample transformations.

```python
import json, ssl, urllib.request, pathlib, base64, hashlib
from cryptography.exceptions import InvalidTag
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.keywrap import aes_key_unwrap
from cryptography.hazmat.primitives.kdf.hkdf import HKDF

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
BASE = "https://retained-archive.arwc.test"

# W12.1: get quarantine to find archive_id and build_id
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/collectors/quarantine", headers=h), context=ctx).read())
rec = d1["record"]
archive_id = rec["archive"]           # ARC-COLD-CRR-2019
build_id = rec["collector_build_id"]  # from quarantine record

# Download AQTN frame
frame = urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/artifacts/QTN-COL-2019-12.aqtn", headers=h),
    context=ctx).read()
# Verify it's AQTN1 format
assert frame[:5] == b"AQTN1"

# Download samples
samples_resp = urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/artifacts/AQTN1-samples.json", headers=h),
    context=ctx).read()
samples_data = json.loads(samples_resp)
samples = samples_data["samples"]

# Derive keys: HKDF(SHA256, length=64, salt=archive_id, info=b'aqtn1/keys').derive(build_id)
keys = HKDF(algorithm=hashes.SHA256(), length=64,
            salt=archive_id.encode(), info=b"aqtn1/keys").derive(build_id.encode())

# Process each sample
out = {}
for s in samples:
    if s["operation"] == "AES-256-GCM decrypt":
        value = AESGCM(keys[:32]).decrypt(
            base64.b64decode(s["nonce"]),
            base64.b64decode(s["ciphertext_and_tag"]),
            base64.b64decode(s["aad"]))
        out[s["sample_id"]] = hashlib.sha256(value).hexdigest()
    elif s["operation"] == "AES-256-KW unwrap":
        value = aes_key_unwrap(keys[32:], base64.b64decode(s["wrapped"]))
        out[s["sample_id"]] = hashlib.sha256(value).hexdigest()
    else:
        try:
            AESGCM(keys[:32]).decrypt(
                base64.b64decode(s["nonce"]),
                base64.b64decode(s["ciphertext_and_tag"]),
                base64.b64decode(s["aad"]))
        except InvalidTag:
            out[s["sample_id"]] = "authentication_failed"

# Submit sample results
body = json.dumps({"sample_results": out}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/bundles/current/collection-policy",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])
print("key_cipher:", d["record"]["key_cipher"])  # AES-256-KW
print("flag:", d["flag"])                        # FLAG{06220fbe7bb08d8d}
```

Flag: `FLAG{06220fbe7bb08d8d}` in the top-level `flag` field.
