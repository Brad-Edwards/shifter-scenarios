Requires W12.2 (protection reconstructed). Open the cold archive. The AQTN1 frame uses the same HKDF key material as the samples in W12.2: `HKDF(SHA256, length=64, salt=archive_id, info=b"aqtn1/keys").derive(build_id)`. The first 32 bytes are the AES-256-GCM key; the last 32 bytes are the AES-256-KW key. Decrypt the frame ciphertext with the GCM key (the metadata is the AAD), then unwrap the PKCS8 private key in `collection["wrapped_pkcs8"]` with the KW key.

```python
import json, ssl, urllib.request, pathlib, base64
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.keywrap import aes_key_unwrap
from cryptography.hazmat.primitives.kdf.hkdf import HKDF

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
BASE = "https://retained-archive.arwc.test"

frame = urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/artifacts/QTN-COL-2019-12.aqtn", headers=h), context=ctx).read()
assert frame[:5] == b"AQTN1"
metadata_length = int.from_bytes(frame[5:9], "big")
build_length = int.from_bytes(frame[49:51], "big")
build_id = frame[51:51 + build_length].decode()       # COLLECT-CRR-12
nonce_start = 51 + build_length
nonce = frame[nonce_start:nonce_start + 12]
metadata_start = nonce_start + 12
metadata = frame[metadata_start:metadata_start + metadata_length]
ciphertext = frame[metadata_start + metadata_length:]

archive_id = "ARC-COLD-CRR-2019"  # from the W12.1 quarantine record
keys = HKDF(algorithm=hashes.SHA256(), length=64,
            salt=archive_id.encode(), info=b"aqtn1/keys").derive(build_id.encode())
plaintext = AESGCM(keys[:32]).decrypt(nonce, ciphertext, metadata)
collection = json.loads(plaintext)
pkcs8_key = aes_key_unwrap(keys[32:], base64.b64decode(collection["wrapped_pkcs8"]))

body = json.dumps({
    "plaintext": base64.b64encode(plaintext).decode(),
    "unwrapped_pkcs8": base64.b64encode(pkcs8_key).decode(),
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/archive-jobs", data=body, headers=h, method="POST"),
    context=ctx).read())
print("record_id:", d["record"]["record_id"])   # ARC-COLD-CRR-2019
print("flag:", d["flag"])                        # FLAG{7b143e6dc66add3e}

# Save for W12.4.
pathlib.Path("receipts/cold-archive-collection.json").write_text(json.dumps(collection))
pathlib.Path("receipts/historical-key.pkcs8.b64").write_text(base64.b64encode(pkcs8_key).decode())
```
