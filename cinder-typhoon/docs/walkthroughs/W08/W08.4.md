Requires W08.3 (collector reconstructed). Decrypt the collector config using HKDF, then submit plaintext.

```python
import json, ssl, urllib.request, pathlib, base64
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.kdf.hkdf import HKDF

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
hg = {k: v for k, v in h.items() if k != "Content-Type"}
BASE = "https://retained-archive.arwc.test"

# Get key material from quarantine and bundle records
d_q = json.loads(urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/collectors/quarantine", headers=hg), context=ctx).read())
archive_id = d_q["record"]["archive"]          # ARC-COLD-CRR-2019
build_id = d_q["record"]["collector_build_id"] # COLLECT-CRR-12

d_b = json.loads(urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/bundles/current", headers=hg), context=ctx).read())
collection_id = d_b["record"]["collection_id"]  # COL-CRR-2019-12
salt = collection_id.rsplit("-", 1)[0].encode()  # b"COL-CRR-2019"

# Derive key: HKDF(SHA256, length=32, salt=collection_id_prefix, info=b"collector-config/v3").derive(build_id)
key = HKDF(algorithm=hashes.SHA256(), length=32, salt=salt,
            info=b"collector-config/v3").derive(build_id.encode())

# Download and decrypt the collector config envelope
envelope = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/artifacts/collector/config", headers=hg),
    context=ctx).read())
nonce = base64.b64decode(envelope["nonce"])
ciphertext_and_tag = base64.b64decode(envelope["ciphertext_and_tag"])
aad = envelope["aad"].encode()
plaintext_text = AESGCM(key).decrypt(nonce, ciphertext_and_tag, aad).decode()
print("Decrypted:", plaintext_text[:30], "...")

# Submit plaintext as TEXT (not base64)
body = json.dumps({"plaintext": plaintext_text}).encode()
d = json.loads(urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/collection-artifacts/current",
    data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # CFG-COLLECT-CRR-12
print("flag:", d["flag"])                        # FLAG{735f3e46ccc090a8}
```

Flag: `FLAG{735f3e46ccc090a8}` in the top-level `flag` field.
