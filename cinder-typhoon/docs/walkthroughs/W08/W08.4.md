Requires W08.3 (collector reconstructed). Decrypt the collector config using HKDF, then submit plaintext.

```python
import json, ssl, urllib.request, pathlib, base64
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.kdf.hkdf import HKDF

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
BASE = "https://retained-archive.arwc.test"

# Get collector config (encrypted envelope)
envelope_bytes = urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/artifacts/collector/config", headers=h),
    context=ctx).read()
envelope = json.loads(envelope_bytes)

# Collect key material
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/bundles/current", headers=h), context=ctx).read())
collection_id = d1["record"]["collection_id"]  # e.g. "COL-CRR-R4"
salt = collection_id.rsplit("-", 1)[0].encode()   # non-revision prefix

# Build ID from W08.3 artifact
build_id = "COLLECT-CRR-12"  # from collector record

# Derive key
key = HKDF(algorithm=hashes.SHA256(), length=32, salt=salt,
            info=b"collector-config/v3").derive(build_id.encode())

# Decrypt
nonce = base64.b64decode(envelope["nonce"])
ciphertext_and_tag = base64.b64decode(envelope["ciphertext_and_tag"])
aad = envelope["aad"].encode()
plaintext = AESGCM(key).decrypt(nonce, ciphertext_and_tag, aad)
print("plaintext:", plaintext.decode()[:30], "...")

# Submit to archive
body = json.dumps({"plaintext": base64.b64encode(plaintext).decode()}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/collection-artifacts/current",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("flag:", d["flag"])   # FLAG{735f3e46ccc090a8}
```

Flag: `FLAG{735f3e46ccc090a8}` in the top-level `flag` field.
