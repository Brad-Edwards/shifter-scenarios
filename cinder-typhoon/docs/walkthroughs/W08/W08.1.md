import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}

d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/bundles/current", headers=h),
    context=ctx).read())
rec = d["record"]
print("record_id:", rec["record_id"])        # BND-COLLECT-CRR-12
print("collection_id:", rec["collection_id"])
print("flag:", d["flag"])                    # FLAG{8243ed9c4cc055e1}
