Requires W08.2 (exchange reconstructed). Download collector parts from archive and submit SHA256 proof.

```python
import json, ssl, urllib.request, pathlib, hashlib, base64

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
BASE = "https://retained-archive.arwc.test"

# W08.1: get bundle to find collector info
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/bundles/current", headers=h), context=ctx).read())
rec = d1["record"]
# (Use collector fragments from exchange reconstruction)

# Download collector parts
fragments = []
for i, expected_sha256 in enumerate(rec.get("fragments", [])):
    part = urllib.request.urlopen(
        urllib.request.Request(f"{BASE}/api/artifacts/collector/part-{i:02d}", headers=h),
        context=ctx).read()
    fragments.append(part)

artifact = b"".join(fragments)
sha256 = hashlib.sha256(artifact).hexdigest()

# Save the artifact
pathlib.Path("retained-artifact").write_bytes(artifact)
pathlib.Path("retained-artifact").chmod(0o700)

# Submit SHA256 proof
body = json.dumps({
    "build_id": rec.get("build_id"),
    "collector_sha256": sha256,
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/collectors/handover",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])
print("flag:", d["flag"])   # FLAG{debd8573ad911cbb}
```

Flag: `FLAG{debd8573ad911cbb}` in the top-level `flag` field.
