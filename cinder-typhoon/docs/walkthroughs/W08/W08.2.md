```python
import json, ssl, urllib.request, pathlib, hashlib, io, tarfile, base64

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
hg = {k: v for k, v in h.items() if k != "Content-Type"}
BASE = "https://retained-archive.arwc.test"

# GF(256) arithmetic for RS(14,12) reconstruction
def gf_multiply(left, right):
    product = 0
    while right:
        if right & 1: product ^= left
        right >>= 1; left <<= 1
        if left & 0x100: left ^= 0x11D
    return product

def gf_power(value, exponent):
    result = 1
    for _ in range(exponent): result = gf_multiply(result, value)
    return result

def reconstruct(retained):
    coefficients = {row: [gf_power(2, row * col) for col in range(12)] for row in (12, 13)}
    residuals = {12: bytearray(retained[12]), 13: bytearray(retained[13])}
    for row in (12, 13):
        for col in range(12):
            if col in (7, 11): continue
            for off, val in enumerate(retained[col]):
                residuals[row][off] ^= gf_multiply(coefficients[row][col], val)
    a, b = coefficients[12][7], coefficients[12][11]
    c, d = coefficients[13][7], coefficients[13][11]
    det = gf_multiply(a, d) ^ gf_multiply(b, c)
    inv_det = gf_power(det, 254)
    s7, s11 = bytearray(96), bytearray(96)
    for off in range(96):
        e, f = residuals[12][off], residuals[13][off]
        s7[off] = gf_multiply(gf_multiply(d, e) ^ gf_multiply(b, f), inv_det)
        s11[off] = gf_multiply(gf_multiply(c, e) ^ gf_multiply(a, f), inv_det)
    return bytes(s7), bytes(s11)

# W08.1: get bundle to find download path
d1 = json.loads(urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/bundles/current", headers=hg), context=ctx).read())
bundle_id = d1["record"]["record_id"]  # BND-COLLECT-CRR-12

# Download artifact (13.9MB tar.gz with exchange shards)
artifact = urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/artifacts/{bundle_id}.tar.gz", headers=hg), context=ctx).read()

with tarfile.open(fileobj=io.BytesIO(artifact), mode="r:gz") as arc:
    manifest = json.loads(arc.extractfile("exchange/manifest.json").read())
    retained = {}
    for i in range(14):
        if i in (7, 11): continue  # withheld
        shard = arc.extractfile(f"exchange/shard-{i:02d}.bin").read()
        retained[i] = shard

shard_7, shard_11 = reconstruct(retained)
assert hashlib.sha256(shard_7).hexdigest() == manifest["sha256"]["7"]
assert hashlib.sha256(shard_11).hexdigest() == manifest["sha256"]["11"]

# Submit reconstructed shards
body = json.dumps({
    "shard_7": base64.b64encode(shard_7).decode(),
    "shard_11": base64.b64encode(shard_11).decode(),
}).encode()
d = json.loads(urllib.request.urlopen(urllib.request.Request(f"{BASE}/api/replay-exchange/current",
    data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # XCHG-COL-CRR-2019-12
print("flag:", d["flag"])                        # FLAG{bef107115ff53d4a}
```
