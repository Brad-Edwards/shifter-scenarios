import json, ssl, urllib.request, pathlib, hashlib, io, tarfile

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}
BASE = "https://retained-archive.arwc.test"

# W08.1: get bundle to find download path
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/bundles/current", headers=h), context=ctx).read())
bundle_id = d1["record"]["record_id"]  # BND-COLLECT-CRR-12

# Download the bundle artifact
artifact_bytes = urllib.request.urlopen(
    urllib.request.Request(f"{BASE}/api/artifacts/{bundle_id}.tar.gz", headers=h),
    context=ctx).read()
print("content_type: application/gzip")
print("size:", len(artifact_bytes))

# Parse the archive and reconstruct missing shards
with tarfile.open(fileobj=io.BytesIO(artifact_bytes), mode="r:gz") as arc:
    manifest = json.loads(arc.extractfile("exchange/manifest.json").read())
    retained = {}
    for index in range(14):
        if index in (7, 11):
            continue  # withheld
        shard = arc.extractfile(f"exchange/shard-{index:02d}.bin").read()
        retained[index] = shard
