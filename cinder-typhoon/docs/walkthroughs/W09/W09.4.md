Requires W09.3 (reserve reconciliation via Metabase). Get integration session from business workplace.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
planner = pathlib.Path("receipts/planner-session").read_text().strip()

# Get integration session from business-workplace
# (discover business.integration-contract route first)
# integration_session comes from: GET business-workplace.arwc.test/api/integration-contracts/current

# Submit integration session to get process_session
integration_session = "<from business-workplace API>"
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc",
     "X-ARWC-Planner-Session": planner,
     "X-ARWC-Integration-Session": integration_session,
     "Content-Type": "application/json"}
body = json.dumps({
    "integration_client": "INT-CRR-08",
    "report_binding": "REC-CRR-DP3-17",
    "scope": "OG-CRR-02/read",
    "feed": "FEED-OG2-R21",
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://planning-data.arwc.test/api/integration-sessions",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("record_id:", d["record"]["record_id"])   # FEED-OG2-R21
print("process_session:", d["process_session"][:8], "...")
print("process_view:", d["process_view"])        # https://process-view.arwc.test
print("flag:", d["flag"])                        # FLAG{2085674d8d2c8b96}

pathlib.Path("receipts").mkdir(exist_ok=True)
pathlib.Path("receipts/process-session").write_text(d["process_session"])
```

Flag: `FLAG{2085674d8d2c8b96}` in the top-level `flag` field.
