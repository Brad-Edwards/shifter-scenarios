## Reconstruct the manifest offline (K03.2 acceptance check)

Decode each stage without executing it: `payload` (base64+zlib) is the PowerShell stage; it carries `$Stage2` (base64+zlib) which is the JavaScript stage; the JavaScript carries a base64 blob and its hex XOR key, which decode to a zlib-compressed JSON manifest.

```
python3 - <<'PY'
import zipfile, base64, zlib, json, re
with zipfile.ZipFile("/home/rowan/work/analysis/field-notes-2026-08.odt") as z:
    stage = z.read("Attachments/staging-FKCOL-2841.bin").decode()
payload_b64 = re.search(r"^payload=(.*)$", stage, re.M).group(1)
powershell = zlib.decompress(base64.b64decode(payload_b64))            # stage 2
stage2_b64 = re.search(r"\$Stage2 = '([^']+)'", powershell.decode()).group(1)
javascript = zlib.decompress(base64.b64decode(stage2_b64))             # stage 3
encoded = re.search(r'const encoded = "([^"]+)"', javascript.decode()).group(1)
key_hex = re.search(r'Buffer\.from\("([0-9a-f]+)", "hex"\)', javascript.decode()).group(1)
key = bytes.fromhex(key_hex)
cipher = base64.b64decode(encoded)
compressed = bytes(v ^ key[i % len(key)] for i, v in enumerate(cipher))
manifest = json.loads(zlib.decompress(compressed))
print(json.dumps(manifest, indent=2))
PY
```
