Requires W16.2 (boundary observed). Submit verify operation with known reference values.

The archive service expects EXACT values:
- reference_values: {"4": 0, "5": 1, "6": 255}
- target_slot: 159 (boundary + 127 = 32 + 127)
- recovered_byte: 0xA5 = 165
- repetitions: 8, bit_order: "lsb0"

```python
import json, ssl, urllib.request, pathlib, uuid

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

d0 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/sequencer-specs/current",
                            headers={"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}),
    context=ctx).read())
rec = d0["record"]; seq_id = rec["record_id"]; revision = rec["revision"]

def binding(**extra):
    return {"sequencer": seq_id, "revision": revision, "request_id": "req-" + uuid.uuid4().hex, **extra}

# Verify with correct values
body = json.dumps(binding(
    operation="verify",
    reference_values={"4": 0, "5": 1, "6": 255},
    target_slot=159,
    recovered_byte=0xA5,
    repetitions=8,
    bit_order="lsb0",
)).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/signal-experiments",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("record_id:", d["record"]["record_id"])         # SEQ-DISCRIM-CRR-4
print("independent_conditions:", d["record"]["independent_conditions"])  # 8
print("flag:", d["flag"])                              # FLAG{ca8e05b4b9107207}
```
