Requires W14.2 (manifest observed). Send CBOR-encoded reading request using process-read/v2 protocol.

```python
import json, ssl, urllib.request, pathlib, time

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
contractor = pathlib.Path("receipts/contractor-session").read_text().strip()

def cbor_text(value):
    encoded = value.encode()
    if len(encoded) < 24: return bytes([0x60 + len(encoded)]) + encoded
    if len(encoded) < 256: return b"\x78" + bytes([len(encoded)]) + encoded
    raise ValueError("text too long")

def cbor_map(value):
    return bytes([0xA0 + len(value)]) + b"".join(cbor_text(k) + cbor_text(v) for k, v in value.items())

observed = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
payload = {
    "asset": "AST-CRR-017",
    "group": "OG-CRR-02",
    "observed_at": observed,
    "nonce": "journey-reading-1",
}
body = cbor_map(payload)

h = {"Authorization": "Bearer " + contractor, "X-ARWC-Tenant": "arwc",
     "Content-Type": "application/cbor", "Content-Length": str(len(body))}
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://field-gateway.arwc.test/api/readings/current",
                            data=body, headers=h, method="POST"),
    context=ctx).read())
print("record_id:", d["record"]["record_id"])   # FEED-OG2-R21
print("process_session:", d["process_session"][:8], "...")
print("flag:", d["flag"])                        # FLAG{027331f36779f480}

pathlib.Path("receipts").mkdir(exist_ok=True)
pathlib.Path("receipts/process-session").write_text(d["process_session"])
```
