## Step 1: Export the diagnostic authority history

```python
import ssl, urllib.request, pathlib, json

# From W31.3 response record
execution_id = "DVL-EXEC-<uuid>"
correlation  = "<uuid>"
history_id   = "HIST-APR-CRR-09"
history_digest = "<sha256>"

token = pathlib.Path("handover/corporate-session").read_text().strip()
context = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
body = json.dumps({
    "execution_id": execution_id,
    "correlation":  correlation,
    "history_id":   history_id,
    "history_digest": history_digest
}).encode()
req = urllib.request.Request(
    "https://process-view.arwc.test/api/diagnostic-programs/authority",
    data=body,
    headers={"Authorization": "Bearer " + token, "X-ARWC-Tenant": "arwc",
             "Content-Type": "application/json"},
    method="POST")
resp = urllib.request.urlopen(req, context=context)
result = json.loads(resp.read())
print(json.dumps(result, indent=2))
```
