Navigate from the workbench index to the dispatch desk, then to the delivery notice:

```bash
curl -s http://workbench.training:8080/ | grep -o 'http://workbench.training:8081[^)]*'
# -> http://workbench.training:8081/
curl -s http://workbench.training:8081/ | grep -o '/notice-[^)]*'
# -> /notice-DL-204.md
curl -s http://workbench.training:8081/notice-DL-204.md | grep webmanifest
# -> `/site.webmanifest`
curl -s http://workbench.training:8081/site.webmanifest | python3 -c "import sys,json; d=json.load(sys.stdin); print([p for p in d['content_indexes'] if 'retired' in p][0])"
# -> /content/retired.json
curl -s http://workbench.training:8081/content/retired.json | python3 -c "import sys,json; d=json.load(sys.stdin); print(d['records'][0]['path'])"
# -> /retired/amendment-AMEND-204.json
curl -s http://workbench.training:8081/retired/amendment-AMEND-204.json
```
