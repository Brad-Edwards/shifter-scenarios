## Recover the sealed viewer's bytecode VM

The sealed viewer runs a small 6-byte-per-instruction VM. Recover it by
disassembly and reimplement it once in Python. Save it; W22.2 and W22.3 reuse it.

```bash
cat > /tmp/w22vm.py <<'PY'
import struct, hashlib
def run_vm(program):
    reg = [0, 0, 0, 0]
    mem = bytearray(256)
    zero = False
    pc = steps = 0
    while 0 <= pc < len(program) // 6 and steps < 4096:
        op, dst, src, imm = struct.unpack_from("<HBBH", program, pc * 6)
        steps += 1
        if op == 1:   reg[dst] = imm; zero = reg[dst] == 0; pc += 1
        elif op == 2: reg[dst] = mem[(reg[src] + imm) & 255]; zero = reg[dst] == 0; pc += 1
        elif op == 3: reg[dst] = (reg[dst] ^ reg[src] ^ imm) & 0xffff; zero = reg[dst] == 0; pc += 1
        elif op == 4: reg[dst] = (reg[dst] + reg[src] + imm) & 0xffff; zero = reg[dst] == 0; pc += 1
        elif op == 5:
            s = imm & 15; v = reg[src]
            reg[dst] = v if not s else ((v << s) | (v >> (16 - s))) & 0xffff
            zero = reg[dst] == 0; pc += 1
        elif op == 6: mem[(reg[dst] + imm) & 255] = reg[src] & 255; pc += 1
        elif op == 7: pc = pc + struct.unpack("<h", struct.pack("<H", imm))[0] if not zero else pc + 1
        elif op == 255: break
        else: raise SystemExit(f"bad opcode {op}")
    return reg, bytes(mem), steps
def public(program):
    reg, mem, steps = run_vm(program)
    return {"registers": reg, "memory_40_80_sha256": hashlib.sha256(mem[0x40:0x80]).hexdigest(), "steps": steps}
PY
```

## Read the sealed viewer delivery

```bash
PROCESS=$(cat receipts/process-session)
curl -sk --cacert /tmp/arwc-ca.crt -H "Authorization: Bearer $PROCESS" -H "X-ARWC-Tenant: arwc" \
  https://process-view.arwc.test/api/sealed-projects/current/viewer -o /tmp/w22.json
python3 - <<'PY'
import json, base64, hashlib, sys
sys.path.insert(0, "/tmp"); from w22vm import run_vm
d = json.load(open("/tmp/w22.json"))
print("flag:", d["flag"])
print("project_id:", d["record"]["project_id"])
viewer = base64.b64decode(d["viewer_b64"], validate=True)
program = base64.b64decode(d["program_b64"], validate=True)
assert viewer.startswith(b"\x7fELF\x02\x01\x01")
assert hashlib.sha256(viewer).hexdigest() == d["record"]["viewer_sha256"]
assert hashlib.sha256(program).hexdigest() == d["record"]["program_sha256"]
reg, mem, steps = run_vm(program)
print("registers:", reg, "steps:", steps)
PY
```

Expected: `flag: FLAG{f8a5c9e2b7fb06db}`, `project_id: CRR-SEALED-PKG-R7`. The
reimplemented VM registers and steps match the viewer's own output. The flag is
in the top-level `flag` field. Keep `/tmp/w22.json` and `/tmp/w22vm.py`.
