## Build program model from binary (K11.1 step)

```python
import base64, hashlib, json, re
binary = open('/tmp/journey-pc/bin/fieldkest-policyc','rb').read()
recovered = {f"0x{int(v,16):02x}": n.decode() for v,n in re.findall(rb"fieldkest_opcode_([0-9a-f]{2})_([A-Z_]+)\x00", binary)}
opcode_model = {"schema":"fieldkest.policy-opcode-model/v1","instruction_width":4,"opcodes":recovered}
start = binary.find(b"FKPC")
offset = int.from_bytes(binary[start+4:start+8],"little")
count = int.from_bytes(binary[start+8:start+10],"little")
width = int.from_bytes(binary[start+10:start+12],"little")
table = binary[start:start+offset+count*width]
program_model = {"schema":"fieldkest.policy-program-model/v1","program_id":"FKPC-POLICY-2026-09-R4","instruction_width":width,"instruction_count":count,"program_base64":base64.b64encode(table).decode(),"program_sha256":hashlib.sha256(table).hexdigest(),"opcode_model":opcode_model}
json.dump(program_model, open('/tmp/program_model.json','w'))
```
