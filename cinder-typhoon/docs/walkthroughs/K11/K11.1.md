## Recover the opcode model and program table from the binary

The binary is not stripped. It carries `fieldkest_opcode_XX_NAME` symbols and a
68-byte `FKPC` program table (12-byte header plus fourteen 4-byte instructions).

```
python3 - <<'PY'
import base64, hashlib, json, re
b = open('/tmp/journey-pc/bin/fieldkest-policyc', 'rb').read()
opcodes = {f"0x{int(v,16):02x}": n.decode() for v, n in re.findall(rb"fieldkest_opcode_([0-9a-f]{2})_([A-Z_]+)\x00", b)}
opcode_model = {"schema": "fieldkest.policy-opcode-model/v1", "instruction_width": 4, "opcodes": opcodes}
s = b.find(b"FKPC")
off = int.from_bytes(b[s+4:s+8], "little"); cnt = int.from_bytes(b[s+8:s+10], "little"); w = int.from_bytes(b[s+10:s+12], "little")
table = b[s:s+off+cnt*w]
program_model = {"schema": "fieldkest.policy-program-model/v1", "program_id": "FKPC-POLICY-2026-09-R4",
    "instruction_width": w, "instruction_count": cnt, "program_base64": base64.b64encode(table).decode(),
    "program_sha256": hashlib.sha256(table).hexdigest(), "opcode_model": opcode_model}
json.dump(program_model, open('/tmp/program_model.json', 'w'))
json.dump(opcode_model, open('/tmp/opcode_model.json', 'w'))
print("program_sha256", program_model["program_sha256"])
PY
```

## Assemble the four fragments

Each fragment is a minimal `FKPC` program that tests one condition and ends with
RETURN. The README Fragments table and the recovered opcodes give the encoding.

```
python3 - <<'PY'
import base64, hashlib, json, os
opcode_model = json.load(open('/tmp/opcode_model.json'))
def ins(op, operand): return bytes([op, 0]) + operand.to_bytes(2, "little")
def table(items): return b"FKPC" + (12).to_bytes(4,"little") + len(items).to_bytes(2,"little") + (4).to_bytes(2,"little") + b"".join(items)
fragments = {
    "tenant-state": ("FKPC-FRAGMENT-TENANT-STATE", [ins(0x00,0), ins(0x04,0), ins(0x07,0)]),
    "connector-version": ("FKPC-FRAGMENT-CONNECTOR-VERSION", [ins(0x00,1), ins(0x01,0), ins(0x03,0), ins(0x07,0)]),
    "signer-lineage": ("FKPC-FRAGMENT-SIGNER-LINEAGE", [ins(0x00,2), ins(0x04,1), ins(0x07,0)]),
    "compatibility-exception": ("FKPC-FRAGMENT-COMPATIBILITY-EXCEPTION", [ins(0x00,3), ins(0x01,1), ins(0x02,0), ins(0x07,0)]),
}
os.makedirs('/tmp/fragments', exist_ok=True)
for fid, (pid, items) in fragments.items():
    t = table(items)
    pm = {"schema": "fieldkest.policy-program-model/v1", "program_id": pid, "instruction_width": 4,
        "instruction_count": len(items), "program_base64": base64.b64encode(t).decode(),
        "program_sha256": hashlib.sha256(t).hexdigest(), "opcode_model": opcode_model}
    json.dump({"operation": "fragment-check", "program_model": pm, "policy_input": {"fragment_id": fid}}, open(f'/tmp/fragments/{fid}.json', 'w'))
PY
```

## Submit each fragment for validation

The service runs each submitted fragment against its server-side validation
cases. A correctly assembled fragment passes every case.

```
for f in tenant-state connector-version signer-lineage compatibility-exception; do
  curl -sS --cacert $CA -u rowan.ito:$TOKEN -X POST -H "Content-Type: application/json" \
    --data-binary @/tmp/fragments/$f.json $EVAL
  echo
done
```

Expected output for each fragment:

```json
{"schema":"fieldkest.policy-migration-result/v1","operation":"fragment-check","fragment_id":"<name>","program_id":"FKPC-FRAGMENT-<NAME>","receipt_id":"PCMIG-...","program_sha256":"...","flag":"FLAG{5f3e067c00b70ca5}"}
```

Gate: a fragment that returns the wrong result for any case fails with `fragment_validation_failed`.
