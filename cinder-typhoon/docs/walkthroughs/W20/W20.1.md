  -H "Authorization: Bearer $PROCESS" \

  https://process-view.arwc.test/api/firmware-images/current | python3 -c "
import sys,json,base64,hashlib,struct
d=json.load(sys.stdin)
image=base64.b64decode(d['image_b64'])
print('flag:', d['flag'])
print('image_sha256:', hashlib.sha256(image).hexdigest())
target=image[64:128]
payload_len=struct.unpack_from('<H',target,8)[0]
rec,asset,date=target[12:12+payload_len].decode().split('\x00')
print('record_id:', rec, 'asset:', asset)
"
