Query the historian for the largest acquisition gap in FlowBM3s Good-quality samples over the last 24 hours, then submit the gap boundaries.

```python
import json, ssl, urllib.request, pathlib, psycopg2, psycopg2.extras, datetime

UTC = datetime.timezone.utc
ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc", "Content-Type": "application/json"}

conn = psycopg2.connect(host="172.29.63.22", port=5432, dbname="historian",
    user="reader", password="yuVxToCIL3DxUeklDAbsjKWbAc8rs8pw", connect_timeout=5)
conn.autocommit = True
with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
    cur.execute("""
        SELECT source_time, collected_time, ingested_at, value, unit,
               transport_quality, process_quality
        FROM process_history
        WHERE tag = 'flow_b_m3s'
          AND source_time >= current_timestamp - interval '24 hours'
          AND process_quality = 'Good'
        ORDER BY source_time
    """)
    rows = cur.fetchall()
conn.close()

# Find the largest source-time gap between consecutive Good samples.
pairs = [(rows[i], rows[i+1]) for i in range(len(rows) - 1)]
before, after = max(pairs, key=lambda p: p[1]["source_time"] - p[0]["source_time"])

gap_seconds = (after["source_time"] - before["source_time"]).total_seconds()

def fmt(ts):
    # Truncate to second (endpoint requires second precision).
    return ts.replace(microsecond=0).astimezone(UTC).isoformat()

body = json.dumps({
    "card": "W18.3",
    "variable": "FlowBM3s",
    "before_source_time": fmt(before["source_time"]),
    "before_collected_time": fmt(before["collected_time"]),
    "before_ingested_at": fmt(before["ingested_at"]),
    "after_source_time": fmt(after["source_time"]),
    "after_collected_time": fmt(after["collected_time"]),
    "after_ingested_at": fmt(after["ingested_at"]),
    "gap_seconds": int(round(gap_seconds)),
    "before_transport_quality": before["transport_quality"],
    "before_process_quality": before["process_quality"],
    "after_transport_quality": after["transport_quality"],
    "after_process_quality": after["process_quality"],
    "unit": before["unit"],
    "interpretation": "source_gap_with_unchanged_engineering_unit",
}).encode()

d = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://process-view.arwc.test/api/live-observations/current",
                           data=body, headers=h, method="POST"),
    context=ctx).read())
print("satisfied:", d["satisfied"])
print("flag:", d["flag"])   # FLAG{d67d3828f16f64ad}
```

The service builds its own product snapshot (last 24 hours) and finds the same largest gap.
All timestamp fields must be truncated to the second; gap_seconds must equal the integer second difference.

Flag: `FLAG{d67d3828f16f64ad}` in the top-level `flag` field.
