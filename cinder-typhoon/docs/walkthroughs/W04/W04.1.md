```python
import json, ssl, urllib.request, pathlib

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca.crt")
sess = pathlib.Path("handover/corporate-session").read_text().strip()
h = {"Authorization": "Bearer " + sess, "X-ARWC-Tenant": "arwc"}

# Get client doc to find archive.contract route
cl = json.loads(urllib.request.urlopen(
    urllib.request.Request("https://retained-archive.arwc.test/.well-known/arwc-client", headers=h),
    context=ctx).read())
contract_url = next(r["template"] for r in cl["routes"] if r["key"] == "archive.contract")
print("contract_url:", contract_url)
# https://retained-archive.arwc.test/api/archive-contracts/current

d = json.loads(urllib.request.urlopen(
    urllib.request.Request(contract_url, headers=h), context=ctx).read())
rec = d["record"]
print("record_id:", rec["record_id"])        # AWF-CRR-229-R4
print("enrollment_token:", rec["enrollment_token"][:8], "...")
print("flag:", rec["flag"])                   # FLAG{57f6312a585e7e4b}

# Save enrollment token for W04.2
pathlib.Path("receipts").mkdir(exist_ok=True)
pathlib.Path("receipts/archive-token").write_text(rec["enrollment_token"])
```

Flag: `FLAG{57f6312a585e7e4b}` in the `flag` field of the archive contract response.
