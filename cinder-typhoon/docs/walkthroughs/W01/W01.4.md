Step 0: get the supplier portal credentials from the W01.1 handover.

```python
import json, ssl, urllib.request, urllib.parse
from http.cookiejar import CookieJar
from urllib.request import HTTPCookieProcessor, HTTPSHandler, build_opener, Request

ctx = ssl.create_default_context(cafile="/tmp/arwc-ca-bundle.pem")
SESSION = open("handover/corporate-session").read().strip()

# W01.1: get handover to find supplier credentials
handover_resp = urllib.request.urlopen(
    Request("https://customer-handover.arwc.test:8443/api/handovers/current",
            headers={"Authorization": "Bearer " + SESSION, "X-ARWC-Tenant": "arwc"}),
    context=ctx)
handover = json.loads(handover_resp.read())
onboarding = handover["record"]["supplier_onboarding"]
NC_USER = onboarding["username"]   # jules.orswick@veybridge.test
NC_PASS = onboarding["password"]
```

Log in to the supplier portal and find the Task:

```python
BASE = "https://suppliers.arwc.test"
opener = build_opener(HTTPCookieProcessor(CookieJar()), HTTPSHandler(context=ctx))
opener.open(Request(f"{BASE}/api/method/login",
    data=urllib.parse.urlencode({"usr": NC_USER, "pwd": NC_PASS}).encode(),
    method="POST"), timeout=30)

subj = "Supplier handback — MTN-CRR-204"
q = urllib.parse.urlencode({
    "filters": json.dumps({"subject": subj}),
    "fields": json.dumps(["name", "status", "description"]),
    "limit_page_length": 2
})
tasks = json.loads(opener.open(Request(f"{BASE}/api/resource/Task?{q}"),
                                timeout=30).read())["data"]
task = tasks[0]
# task["name"] = "TASK-2026-00013", task["status"] = "Open"
```

Append the inspection reference and update the Task status:

```python
desc = (task.get("description") or "").rstrip() + " Confirmed inspection: INSP-CRR-2026-09-18"
put_body = json.dumps({"status": "Working", "description": desc}).encode()
result = json.loads(opener.open(
    Request(f"{BASE}/api/resource/Task/{urllib.parse.quote(task['name'], safe='')}",
            data=put_body, method="PUT",
            headers={"Content-Type": "application/json"}),
    timeout=30).read())["data"]
print("status:", result["status"])         # Working
print("flag:", result["cinder_receipt"])   # FLAG{ce5d42bc2791d4e5}
```
